```python path=tests/conftest.py
"""共享 fixture：TestClient 与错误响应断言辅助。

设计约束（见 docs/architecture.md）：
- 测试不启动真实端口，使用 FastAPI TestClient（AC14 一键 pytest）。
- 每个用例通过函数级 fixture 拿到独立 client，用例之间无状态共享。
"""

import pytest
from fastapi.testclient import TestClient


@pytest.fixture()
def client():
    """函数级 TestClient，保证用例相互独立、可重复执行。"""
    from app.main import app

    with TestClient(app) as c:
        yield c


@pytest.fixture()
def post_add(client):
    """POST /add 的便捷封装，显式指定 application/json。"""

    def _post(payload, **kwargs):
        return client.post("/add", json=payload, **kwargs)

    return _post


def assert_error_body(resp, expected_status, expected_code):
    """统一校验错误响应形状（AC13）：合法 JSON + 非空 error_code/message。"""
    assert resp.status_code == expected_status, (
        f"期望 HTTP {expected_status}，实际 {resp.status_code}，body={resp.text!r}"
    )
    body = resp.json()  # 非法 JSON 会在此抛错，即 AC13 失败
    assert isinstance(body, dict), f"错误响应体必须是 JSON 对象，实际 {type(body)}"
    assert body.get("error_code") == expected_code, (
        f"期望 error_code={expected_code!r}，实际 {body.get('error_code')!r}，body={body}"
    )
    message = body.get("message")
    assert isinstance(message, str) and message.strip(), (
        f"message 必须是非空字符串，实际 {message!r}"
    )
    return body
```

```python path=tests/test_add.py
"""AC1–AC15：POST /add 正常路径、参数校验、解析错误、媒体类型与错误形状。"""

import pytest

from tests.conftest import assert_error_body


# ---------- AC1 / AC2 / AC3：正常路径 ----------

def test_add_integers_returns_three(post_add):
    """AC1：{"a":1,"b":2} → 200，result == 3。"""
    resp = post_add({"a": 1, "b": 2})
    assert resp.status_code == 200, resp.text
    body = resp.json()
    assert body["result"] == 3
    assert isinstance(body["result"], (int, float)) and not isinstance(body["result"], bool)


def test_add_float_and_int_returns_three_point_five(post_add):
    """AC2：{"a":1.5,"b":2} → 200，result == 3.5。"""
    resp = post_add({"a": 1.5, "b": 2})
    assert resp.status_code == 200, resp.text
    assert resp.json()["result"] == 3.5


def test_add_echoes_operands(post_add):
    """AC3：响应回显 a、b，且与请求一致。"""
    resp = post_add({"a": 1.5, "b": 2})
    assert resp.status_code == 200, resp.text
    body = resp.json()
    assert body["a"] == 1.5
    assert body["b"] == 2
    assert set(body) >= {"result", "a", "b"}


# ---------- AC4：负数 / 零 / 小数 ----------

@pytest.mark.parametrize(
    "a,b,expected",
    [
        (-5, 3, -2),
        (0, 0, 0),
        (-1.5, -2.5, -4.0),
    ],
    ids=["negative_plus_positive", "zero_plus_zero", "negative_floats"],
)
def test_add_supports_negative_zero_and_float(post_add, a, b, expected):
    """AC4：负数、零、负小数均返回正确结果。"""
    resp = post_add({"a": a, "b": b})
    assert resp.status_code == 200, resp.text
    assert resp.json()["result"] == expected


# ---------- AC5：IEEE 754 严格精度 ----------

def test_add_float_precision_is_strict_ieee754(post_add):
    """AC5：0.1 + 0.2 严格等于 0.30000000000000004，不做舍入。"""
    resp = post_add({"a": 0.1, "b": 0.2})
    assert resp.status_code == 200, resp.text
    result = resp.json()["result"]
    assert result == 0.30000000000000004, f"实际 {result!r}，疑似做了舍入"
    assert result != 0.3, "结果被舍入为 0.3，违反 AC5"


# ---------- AC6 / AC7：字段缺失 ----------

def test_missing_b_returns_invalid_argument_field_b(post_add):
    """AC6：缺 b → 400 INVALID_ARGUMENT，field == "b"。"""
    resp = post_add({"a": 1})
    body = assert_error_body(resp, 400, "INVALID_ARGUMENT")
    assert body.get("field") == "b", f"field 应为 'b'，实际 {body.get('field')!r}"


def test_missing_a_returns_invalid_argument_field_a(post_add):
    """AC7：缺 a → 400 INVALID_ARGUMENT，field == "a"。"""
    resp = post_add({"b": 1})
    body = assert_error_body(resp, 400, "INVALID_ARGUMENT")
    assert body.get("field") == "a", f"field 应为 'a'，实际 {body.get('field')!r}"


# ---------- AC8 / AC9 / AC10：类型非法 ----------

def test_string_number_rejected(post_add):
    """AC8：a 为字符串数字 "1" → 400 INVALID_ARGUMENT（禁止隐式强转）。"""
    resp = post_add({"a": "1", "b": 2})
    body = assert_error_body(resp, 400, "INVALID_ARGUMENT")
    assert body.get("field") == "a"


def test_boolean_rejected(post_add):
    """AC9：a 为 true → 400 INVALID_ARGUMENT（bool 是 int 子类的陷阱）。"""
    resp = post_add({"a": True, "b": 2})
    body = assert_error_body(resp, 400, "INVALID_ARGUMENT")
    assert body.get("field") == "a"


def test_null_rejected(post_add):
    """AC10：a 为 null → 400 INVALID_ARGUMENT。"""
    resp = post_add({"a": None, "b": 2})
    body = assert_error_body(resp, 400, "INVALID_ARGUMENT")
    assert body.get("field") == "a"


# ---------- AC11：非法 JSON ----------

def test_invalid_json_body_returns_invalid_json(client):
    """AC11：body 为 not-json → 400 INVALID_JSON。"""
    resp = client.post(
        "/add",
        content=b"not-json",
        headers={"Content-Type": "application/json"},
    )
    assert_error_body(resp, 400, "INVALID_JSON")


def test_empty_body_returns_invalid_json(client):
    """AC11 边界：空 body → 400 INVALID_JSON。"""
    resp = client.post(
        "/add",
        content=b"",
        headers={"Content-Type": "application/json"},
    )
    assert_error_body(resp, 400, "INVALID_JSON")


# ---------- AC12：媒体类型 ----------

def test_text_plain_returns_415(client):
    """AC12：Content-Type: text/plain → 415（且优先于 JSON 解析，见 C4）。"""
    resp = client.post(
        "/add",
        content=b"not-json",
        headers={"Content-Type": "text/plain"},
    )
    assert_error_body(resp, 415, "UNSUPPORTED_MEDIA_TYPE")


def test_json_with_charset_is_accepted(client):
    """AC12 边界：application/json; charset=utf-8 应被接受，不得误判 415。"""
    resp = client.post(
        "/add",
        content=b'{"a": 1, "b": 2}',
        headers={"Content-Type": "application/json; charset=utf-8"},
    )
    assert resp.status_code == 200, resp.text
    assert resp.json()["result"] == 3


# ---------- AC13：错误响应形状 ----------

def test_all_error_responses_are_structured_json(client):
    """AC13：各类错误响应体均为合法 JSON，含非空 error_code 与 message。"""
    cases = [
        ({"json": {"a": 1}}, 400, "INVALID_ARGUMENT"),
        (
            {"content": b"not-json", "headers": {"Content-Type": "application/json"}},
            400,
            "INVALID_JSON",
        ),
        (
            {"content": b"x", "headers": {"Content-Type": "text/plain"}},
            415,
            "UNSUPPORTED_MEDIA_TYPE",
        ),
        ({"json": {"a": 1e308, "b": 1e308}}, 422, "RESULT_NOT_FINITE"),
    ]
    for kwargs, status, code in cases:
        resp = client.post("/add", **kwargs)
        assert_error_body(resp, status, code)


def test_json_array_body_returns_invalid_argument(client):
    """AC13 补充：body 为 JSON 数组 → 400 INVALID_ARGUMENT，不得 5xx。"""
    resp = client.post("/add", json=[1, 2])
    assert resp.status_code < 500, f"不得返回 5xx，实际 {resp.status_code}"
    assert_error_body(resp, 400, "INVALID_ARGUMENT")


def test_empty_object_returns_invalid_argument_field_a(post_add):
    """AC13 补充：body 为 {} → 400 INVALID_ARGUMENT，field == "a"。"""
    resp = post_add({})
    body = assert_error_body(resp, 400, "INVALID_ARGUMENT")
    assert body.get("field") == "a"
```
