```python path=app/schemas.py
"""请求/响应实体定义。

设计决策 R2：请求体不使用 pydantic 模型做校验（pydantic 会把 "1" 强转为 1.0、
把 true 转成 1.0，违反 AC8/AC9），请求体在 routes 层用 Request 手动取 JSON 并
自定义校验；本模块只承载**响应**模型与实体字段的文档化定义。
追溯：需求 5 节实体表。
"""

from __future__ import annotations

from typing import List, Optional

from pydantic import BaseModel, Field

from app.errors import ErrorCode


class AddRequest(BaseModel):
    """实体：AddRequest（仅作文档/类型参考，不用于请求校验）。

    字段约束：a、b 均为必填 JSON 数值，不接受字符串/布尔/null。
    """

    a: float = Field(..., description="加数 a，必须是 JSON 数值")
    b: float = Field(..., description="加数 b，必须是 JSON 数值")


class AddResponse(BaseModel):
    """实体：AddResponse（AC1/AC2/AC3）。"""

    result: float = Field(..., description="a + b 的结果")
    a: float = Field(..., description="回显请求中的 a")
    b: float = Field(..., description="回显请求中的 b")


class ErrorResponse(BaseModel):
    """实体：ErrorResponse（AC13）。

    field / index 为可选键，未使用时省略（设计 5.5）。
    """

    error_code: str = Field(..., description="错误码，取值见 ErrorCode")
    message: str = Field(..., description="非空可读说明")
    field: Optional[str] = Field(None, description="参数校验失败时的字段名")
    index: Optional[int] = Field(None, description="批量接口中出错元素下标")


class HealthResponse(BaseModel):
    """实体：HealthResponse（AC16）。"""

    status: str = Field(default="ok", description='固定为 "ok"')


class BatchRequest(BaseModel):
    """实体：BatchRequest（仅作文档/类型参考，不用于请求校验）。

    约束：items 为 array<object>，长度 1–100，每个元素含 a、b。
    """

    items: List[AddRequest] = Field(..., description="待计算的元素数组，长度 1–100")


class BatchResponse(BaseModel):
    """实体：BatchResponse（AC17）。"""

    results: List[float] = Field(..., description="与 items 等长、同序的结果数组")


__all__ = [
    "AddRequest",
    "AddResponse",
    "ErrorResponse",
    "HealthResponse",
    "BatchRequest",
    "BatchResponse",
    "ErrorCode",
]
```

```python path=app/service.py
"""纯计算逻辑层：不含任何 HTTP 概念，不 import FastAPI。

设计决策：AC1–AC5 的算术语义（尤其 AC5 的 IEEE 754 严格相等）可脱离 HTTP 单测；
F7 的有限性检查只有一处实现。
追溯：F1, F3, F6, F7。
"""

from __future__ import annotations

import logging
import math
from typing import Any, Dict, List, Sequence

from app.errors import ApiError, ErrorCode

logger = logging.getLogger(__name__)

#: 批量接口 items 长度上限（需求 4.3）
MAX_BATCH_SIZE = 100


def is_number(value: Any) -> bool:
    """判断是否为合法 JSON 数值。

    陷阱（C5 / AC9）：Python 中 ``isinstance(True, int)`` 为 True，
    因此必须**先**排除 bool，再判断 int/float。
    """
    if isinstance(value, bool):
        return False
    return isinstance(value, (int, float))


def ensure_finite(result: float, *, index: int | None = None) -> float:
    """校验计算结果为有限数（AC19 / F7）。

    ``math.isfinite`` 对 NaN 与 ±Infinity 均返回 False。
    非有限时抛 ``ApiError(RESULT_NOT_FINITE)``，HTTP 状态 422。
    """
    if not math.isfinite(result):
        logger.warning("non-finite result detected: %r (index=%r)", result, index)
        raise ApiError(
            ErrorCode.RESULT_NOT_FINITE,
            "result is not a finite number (NaN or Infinity)",
            index=index,
        )
    return result


def add(a: float, b: float) -> float:
    """返回 a + b。

    严格 IEEE 754 双精度语义（C2 / AC5）：**禁止** round()、Decimal、
    math.isclose，直接相加。
    """
    if not is_number(a):
        raise ApiError(
            ErrorCode.INVALID_ARGUMENT,
            "field 'a' must be a JSON number",
            field="a",
        )
    if not is_number(b):
        raise ApiError(
            ErrorCode.INVALID_ARGUMENT,
            "field 'b' must be a JSON number",
            field="b",
        )
    return a + b


def add_checked(a: float, b: float) -> float:
    """add + 有限性检查的组合，供 routes 层直接使用。"""
    return ensure_finite(add(a, b))


def validate_pair(payload: Any, *, index: int | None = None) -> tuple[float, float]:
    """校验一个 {a, b} 对象并返回 (a, b)。

    校验顺序：对象类型 → 字段存在性/非 null（a 先于 b）→ 数值类型（排除 bool）。
    缺失字段与 null 走同一分支，field 取首个问题字段（AC6/AC7/AC10）。
    """
    if not isinstance(payload, dict):
        raise ApiError(
            ErrorCode.INVALID_ARGUMENT,
            "request body must be a JSON object with fields 'a' and 'b'",
            index=index,
        )

    values: Dict[str, float] = {}
    for name in ("a", "b"):
        if name not in payload:
            raise ApiError(
                ErrorCode.INVALID_ARGUMENT,
                f"field '{name}' is required",
                field=name,
                index=index,
            )
        value = payload[name]
        if value is None:
            raise ApiError(
                ErrorCode.INVALID_ARGUMENT,
                f"field '{name}' must not be null",
                field=name,
                index=index,
            )
        if not is_number(value):
            raise ApiError(
                ErrorCode.INVALID_ARGUMENT,
                f"field '{name}' must be a JSON number",
                field=name,
                index=index,
            )
        values[name] = value

    return values["a"], values["b"]


def add_batch(items: Sequence[Any]) -> List[float]:
    """批量加法（AC17/AC18）。

    逐元素复用同一套校验与 add，保持顺序；首个出错元素即返回，
    index 为 0-based。
    """
    if not isinstance(items, (list, tuple)):
        raise ApiError(
            ErrorCode.INVALID_ARGUMENT,
            "field 'items' must be an array",
            field="items",
        )
    if len(items) < 1 or len(items) > MAX_BATCH_SIZE:
        raise ApiError(
            ErrorCode.INVALID_ARGUMENT,
            f"field 'items' must contain between 1 and {MAX_BATCH_SIZE} elements",
            field="items",
        )

    results: List[float] = []
    for position, item in enumerate(items):
        a, b = validate_pair(item, index=position)
        results.append(ensure_finite(add(a, b), index=position))
    return results


__all__ = [
    "MAX_BATCH_SIZE",
    "is_number",
    "ensure_finite",
    "add",
    "add_checked",
    "validate_pair",
    "add_batch",
]
```

```python path=app/routes.py
"""路由与校验顺序编排。

全局不变量（C4）：Content-Type → JSON 解析 → 字段校验 → 计算 → 有限性检查。
顺序颠倒会让 text/plain 的非 JSON body 误返 400 而非 415（AC11 vs AC12）。
追溯：F1, F2, F5, F6。
"""

from __future__ import annotations

import json
import logging
from typing import Any, Dict

from fastapi import APIRouter, Request
from fastapi.responses import JSONResponse

from app.errors import ApiError, ErrorCode
from app.schemas import AddResponse, BatchResponse, HealthResponse
from app.service import add, add_batch, ensure_finite, validate_pair

logger = logging.getLogger(__name__)

router = APIRouter()

#: 允许的 Content-Type 主类型（允许 application/json; charset=utf-8）
_JSON_MEDIA_TYPE = "application/json"


def _check_content_type(request: Request) -> None:
    """步骤 1：Content-Type 检查（AC12）。

    在 JSON 解析之前执行，保证 text/plain 的非 JSON body 返回 415。
    """
    raw = request.headers.get("content-type")
    if raw is None:
        # 无 Content-Type 视为不支持（需求 4.1 要求 application/json）
        raise ApiError(
            ErrorCode.UNSUPPORTED_MEDIA_TYPE,
            "Content-Type header must be application/json",
        )
    media_type = raw.split(";", 1)[0].strip().lower()
    if media_type != _JSON_MEDIA_TYPE:
        raise ApiError(
            ErrorCode.UNSUPPORTED_MEDIA_TYPE,
            f"unsupported media type '{media_type}', expected application/json",
        )


async def _read_json(request: Request) -> Any:
    """步骤 2：JSON 解析（AC11）。空 body 与非法 JSON 同码。"""
    body = await request.body()
    if not body or not body.strip():
        raise ApiError(ErrorCode.INVALID_JSON, "request body must not be empty")
    try:
        return json.loads(body)
    except (json.JSONDecodeError, UnicodeDecodeError) as exc:
        logger.info("JSON parse failure: %s", exc)
        raise ApiError(
            ErrorCode.INVALID_JSON,
            "request body is not valid JSON",
        ) from exc


@router.post("/add", response_model=AddResponse)
async def add_endpoint(request: Request) -> JSONResponse:
    """POST /add：返回两数之和（AC1–AC13, AC19）。"""
    _check_content_type(request)
    payload = await _read_json(request)

    # 步骤 3：字段存在性 + 类型校验（AC6–AC10）
    a, b = validate_pair(payload)

    # 步骤 4：计算（AC1–AC5，严格 IEEE 754）
    result = add(a, b)

    # 步骤 5：有限性检查（AC19）
    ensure_finite(result)

    # AC3：显式回显 a、b，不用 **payload 展开
    return JSONResponse(
        status_code=200,
        content={"result": result, "a": a, "b": b},
    )


@router.get("/health", response_model=HealthResponse)
async def health_endpoint() -> Dict[str, str]:
    """GET /health：健康检查（AC16）。无入参、无依赖，恒成功。"""
    return {"status": "ok"}


@router.post("/add/batch", response_model=BatchResponse)
async def add_batch_endpoint(request: Request) -> JSONResponse:
    """POST /add/batch：批量加法（AC17/AC18）。"""
    _check_content_type(request)
    payload = await _read_json(request)

    if not isinstance(payload, dict):
        raise ApiError(
            ErrorCode.INVALID_ARGUMENT,
            "request body must be a JSON object with field 'items'",
            field="items",
        )
    if "items" not in payload:
        raise ApiError(
            ErrorCode.INVALID_ARGUMENT,
            "field 'items' is required",
            field="items",
        )

    results = add_batch(payload["items"])
    return JSONResponse(status_code=200, content={"results": results})


__all__ = ["router", "add_endpoint", "health_endpoint", "add_batch_endpoint"]
```

```python path=app/main.py
"""应用装配、异常处理器注册、app 实例导出。

AC13 要求所有错误响应体均为合法 JSON 且含非空 error_code / message，
因此这里注册 ApiError、RequestValidationError 与兜底 Exception 三个处理器，
保证不出现 5xx 裸栈。
追溯：F1, F2, F5。
"""

from __future__ import annotations

import logging

from fastapi import FastAPI, Request
from fastapi.exceptions import RequestValidationError
from fastapi.responses import JSONResponse

from app.errors import ApiError, ErrorCode, error_response
from app.routes import router

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s %(levelname)s %(name)s %(message)s",
)
logger = logging.getLogger(__name__)

app = FastAPI(
    title="Calculator Add API",
    version="1.0.0",
    description="计算器加法 REST API：POST /add、GET /health、POST /add/batch",
)

app.include_router(router)


@app.exception_handler(ApiError)
async def api_error_handler(request: Request, exc: ApiError) -> JSONResponse:
    """业务异常 → 结构化错误响应（AC6–AC12, AC18, AC19）。"""
    logger.info(
        "ApiError on %s %s: %s field=%r index=%r",
        request.method,
        request.url.path,
        exc.error_code,
        exc.field,
        exc.index,
    )
    return JSONResponse(status_code=exc.status_code, content=exc.to_payload())


@app.exception_handler(RequestValidationError)
async def validation_error_handler(
    request: Request, exc: RequestValidationError
) -> JSONResponse:
    """框架级校验异常 → 400 INVALID_ARGUMENT（保持错误形状统一）。"""
    logger.info("RequestValidationError on %s: %s", request.url.path, exc.errors())
    return JSONResponse(
        status_code=400,
        content=error_response(
            ErrorCode.INVALID_ARGUMENT,
            "request validation failed",
        ),
    )


@app.exception_handler(Exception)
async def unhandled_exception_handler(
    request: Request, exc: Exception
) -> JSONResponse:
    """兜底处理器：仍返回合法 JSON，避免裸栈（AC13）。"""
    logger.exception("Unhandled exception on %s %s", request.method, request.url.path)
    return JSONResponse(
        status_code=500,
        content=error_response(
            ErrorCode.INTERNAL_ERROR,
            "internal server error",
        ),
    )


__all__ = ["app"]
```

```python path=tests/test_overflow.py
from __future__ import annotations

from fastapi.testclient import TestClient


def test_add_overflow(client: TestClient) -> None:
```
